function updateGeoDot(now) { if (geoPhase === 'delay') { geoOpacity = 0; if (now - geoPhaseStart >= INITIAL_DELAY_MS) { geoPhase = 'stepping'; geoStepIndex = 0; geoStepStart = now; geoDistance = geoSequence[0].dist; geoOpacity = 1; geoFrozenDirection = compassDirection; } } else if (geoPhase === 'stepping') { const sinceStep = now - geoStepStart; if (sinceStep <= STEP_HOLD_MS) { geoOpacity = 1; } else { const ft = Math.min((sinceStep - STEP_HOLD_MS) / STEP_FADE_MS, 1); geoOpacity = 1 - ft; } const nextStepAt = geoSequence[geoStepIndex].delay; const EARLY_MS = 250; const timeUntilNext = nextStepAt - sinceStep; if (!blipPreFired && timeUntilNext <= EARLY_MS && timeUntilNext > 0) { playBlip(); blipPreFired = true; } if (sinceStep >= geoSequence[geoStepIndex].delay) { geoStepIndex++; geoStepStart = now; blipPreFired = false; if (geoStepIndex < geoSequence.length) { geoDistance = geoSequence[geoStepIndex].dist; geoOpacity = 1; } else { geoPhase = 'pause'; geoPhaseStart = now; geoOpacity = 0; } } } else if (geoPhase === 'pause') { geoOpacity = 0; if (now - geoPhaseStart >= CYCLE_PAUSE_MS) { startGeoCycle(now); } } }
